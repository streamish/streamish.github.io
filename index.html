// type definition generated from OpenAPI YAML using the
// "openapi-typescript" library and converted to typed client 
// with "openapi-fetch" (see openapi-ts.dev)
import createClient from 'openapi-fetch';
import type { paths } from 'src/openapi-schema';

// create typed client for API
const client = createClient<paths>({
  baseUrl: `http://localhost:7000`,
});

// sign in
const { data: sessionData } = await client.POST(`/api/create-session`, {
  body: {
    username: 'woodrow',
    password: '*********',
  },
});

// list albums with tracks
const { data: albumData } = await client.GET(`/api/list-albums-with-tracks`, {
  params: {
    query: {
      artist: 'Michael Jackson'
    },
    header: {
      Authorization: `Bearer ${sessionData?.token}`,
    }
  },
});
albumData?.albums.forEach((album) => {
  console.log(album.title);
  album.tracks?.forEach((track) => {
    console.log(track.title);
  });
});
// Swift: URLSession + Codable
import Foundation

struct Track: Codable { let title: String }
struct Album: Codable { let title: String; let tracks: [Track]? }
struct Albums: Codable { let albums: [Album] }
struct Session: Codable { let token: String }

let baseUrl = URL(string: "http://localhost:7000")!

// sign in
var signIn = URLRequest(url: baseUrl.appendingPathComponent("api/create-session"))
signIn.httpMethod = "POST"
signIn.setValue("application/json", forHTTPHeaderField: "Content-Type")
signIn.httpBody = try JSONEncoder().encode([
  "username": "woodrow",
  "password": "*********",
])
let (sessionBytes, _) = try await URLSession.shared.data(for: signIn)
let session = try JSONDecoder().decode(Session.self, from: sessionBytes)

// list albums with tracks
var components = URLComponents(
  url: baseUrl.appendingPathComponent("api/list-albums-with-tracks"),
  resolvingAgainstBaseURL: false)!
components.queryItems = [URLQueryItem(name: "artist", value: "Michael Jackson")]
var request = URLRequest(url: components.url!)
request.setValue(session.token, forHTTPHeaderField: "Authorization")
let (albumBytes, _) = try await URLSession.shared.data(for: request)
let albumData = try JSONDecoder().decode(Albums.self, from: albumBytes)
for album in albumData.albums {
  print(album.title)
  for track in album.tracks ?? [] {
    print(track.title)
  }
}
// Java: OkHttp + Gson
import okhttp3.*;
import com.google.gson.*;

OkHttpClient client = new OkHttpClient();
Gson gson = new Gson();
String baseUrl = "http://localhost:7000";

// sign in
JsonObject credentials = new JsonObject();
credentials.addProperty("username", "woodrow");
credentials.addProperty("password", "*********");
Request signIn = new Request.Builder()
  .url(baseUrl + "/api/create-session")
  .post(RequestBody.create(credentials.toString(),
    MediaType.get("application/json")))
  .build();
try (Response response = client.newCall(signIn).execute()) {
  JsonObject session = gson.fromJson(response.body().string(), JsonObject.class);
  String token = session.get("token").getAsString();

  // list albums with tracks
  HttpUrl url = HttpUrl.parse(baseUrl + "/api/list-albums-with-tracks")
    .newBuilder()
    .addQueryParameter("artist", "Michael Jackson")
    .build();
  Request request = new Request.Builder()
    .url(url)
    .header("Authorization", token)
    .build();
  try (Response albumResponse = client.newCall(request).execute()) {
    JsonObject albumData = gson.fromJson(albumResponse.body().string(), JsonObject.class);
    for (JsonElement a : albumData.getAsJsonArray("albums")) {
      JsonObject album = a.getAsJsonObject();
      System.out.println(album.get("title").getAsString());
      if (album.has("tracks")) {
        for (JsonElement t : album.getAsJsonArray("tracks")) {
          System.out.println(t.getAsJsonObject().get("title").getAsString());
        }
      }
    }
  }
}
